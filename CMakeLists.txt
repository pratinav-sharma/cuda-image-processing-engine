cmake_minimum_required(VERSION 3.24)
# Use the GPU on the build machine unless the caller explicitly chooses a target.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
project(cuda_image_processing_engine VERSION 0.1.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

add_library(image_processing src/grayscale.cpp src/grayscale.cu src/benchmark.cpp src/blur.cpp)
target_include_directories(image_processing PUBLIC include)
set_target_properties(image_processing PROPERTIES CUDA_RUNTIME_LIBRARY Static)
option(IMAGE_CUDA_LINEINFO "Include CUDA source lines in profiler and sanitizer reports" OFF)
if(IMAGE_CUDA_LINEINFO)
  target_compile_options(image_processing PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()

add_library(image_io src/image_io.cpp)
target_include_directories(image_io PUBLIC include PRIVATE third_party/stb)

add_executable(image_demo src/main.cpp)
target_link_libraries(image_demo PRIVATE image_processing image_io)

add_executable(image_benchmark src/benchmark_main.cpp)
target_link_libraries(image_benchmark PRIVATE image_processing image_io)
target_compile_definitions(image_benchmark PRIVATE IMAGE_BUILD_CONFIG="$<CONFIG>")

include(CTest)
if(BUILD_TESTING)
  add_executable(blur_correctness tests/blur.cpp)
  target_link_libraries(blur_correctness PRIVATE image_processing)
  add_test(NAME box_blur_correctness COMMAND blur_correctness)
  set_tests_properties(box_blur_correctness PROPERTIES TIMEOUT 60)
  add_executable(image_correctness tests/correctness.cpp)
  target_link_libraries(image_correctness PRIVATE image_processing image_io)
  add_test(NAME grayscale_gpu_matches_cpu COMMAND image_correctness "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(grayscale_gpu_matches_cpu PROPERTIES TIMEOUT 60)
  add_executable(image_io_correctness tests/image_io.cpp)
  target_link_libraries(image_io_correctness PRIVATE image_io)
  add_test(NAME image_file_io COMMAND image_io_correctness "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(image_file_io PROPERTIES TIMEOUT 60)
  add_test(NAME image_demo_help COMMAND image_demo --help)
  add_executable(benchmark_correctness tests/benchmark.cpp)
  target_link_libraries(benchmark_correctness PRIVATE image_processing)
  add_test(NAME benchmark_validation COMMAND benchmark_correctness)
  set_tests_properties(benchmark_validation PROPERTIES TIMEOUT 60)
  add_test(NAME image_benchmark_help COMMAND image_benchmark --help)
  add_test(NAME image_benchmark_fixture COMMAND image_benchmark "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/rgb.png" --iterations 3 --warmup 1)
  set_tests_properties(image_benchmark_fixture PROPERTIES TIMEOUT 60)
  foreach(threads IN ITEMS 128 256 512)
    add_test(NAME image_benchmark_threads_${threads}
      COMMAND image_benchmark "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/rgb.png" --threads ${threads} --iterations 2 --warmup 1)
    set_tests_properties(image_benchmark_threads_${threads} PROPERTIES TIMEOUT 60)
  endforeach()
endif()
